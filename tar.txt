struct TarString(size: usize){
    Text: [u8; size-1];
    End: u8;
    assert(End == 0 || End == ' ');
}

#[leftpad('0')]
struct Oct(size: usize = 8) { // default values are allowed
    Digits: TarString(size).Text;
    let value = integer(Digits, 8); // or other name? like int or Int or Integer?
}

#[align(256), pad(0)]
struct Header() {
    FilePathAndName: TarString(100).Text;
    FileMode: Oct().value;
    UID: Oct().value;
    GID: Oct().value;
    FileSize: Oct(12).value;
    LastModified: Oct(12).value;
    CheckSum: Oct().value;
    FileType: Enum<u8>({ // anonymous enum
        '0' | 0 => NormalFile,
        '1' => HardLink,
        '2' => SymbolicLink,
        '3' => CharacterSpecial,
        '4' => BlockSpecial,
        '5' => Directory,
        '6' => FIFO,
        '7' => ContiguousFile,
        'g' => GlobalExtendedHeaderWithMetadata,
        'x' => ExtendedHeaderWithMetadataForTheNextFileInTheArchive,
        'A'..'Z' => VendorSpecificExtentions
        _ => Reserved
    })
    assert(FileType != FileType::Reserved);
    NameOfLinkedFile: TarString(100).Text;
    #[nocheck()]
    Unused: [u8; 255] { // extention, anonymous spec
        UStarIndicator: TarString(6).Text;
        UStarVersion: TarString(2).Text;
        OwnerUserName: TarString(32).Text;
        OwnerGroupName: TarString(32).Text;
        DeviceMajorNumber: Oct.value;
        DeviceMinorNumber: Oct.value;
        FilenamePrefix: TarString(155).Text;
        Unspecified: [u8; 12];
    }
    /* nocheck ignores assert checks and Errors (all subfields also will have this property)
    using ? (like `Unused?` or `Unused.UStarIndicator?`) returns a boolean saying if they are valid (no assert mismatches)
    using ! (like `Unused!` or `Unused.UStarIndicator!`) will use the value, if there were errors, the error will be raised at the place of use. similar to .unwrap()
    */
    assert(
        sum(Header as [u8]) // only read fields are available
        - sum(CheckSum as [u8]) + (CheckSum.size * ' ') // sum of all headers with CheckSum replaced as spaces.
        == CheckSum.value
    );
    // for use outside:
    if Unused.UStarIndicator? {
        let filename = Unused.FilenamePrefix! + FilePathAndName;
    } else {
        let filename = FilePathAndName;
    }
}

spec TarEntry() {
    Header: Header();
    FileData: [u8; Header.FileSize];
}

spec File() {
    Entries: [TarEntry()];
}
